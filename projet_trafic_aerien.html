Analyse du Trafic Aérien

Données ouvertes OpenSky Network · PostgreSQL · Power BI

Projet data complet : concevoir et exploiter une base de données relationnelle pour suivre le trafic aérien en temps réel, à partir de l'API ouverte OpenSky Network.

Les états de vol (position, altitude, vitesse, pays d'origine, statut en vol ou au sol) sont collectés en Python via l'API REST, historisés dans PostgreSQL à chaque relevé, puis analysés en SQL (agrégations, zones géographiques, vues). Les résultats sont restitués sous forme de tableaux de bord interactifs sur Power BI.

Code Python — Collecte

collecte_opensky.py

import requests
import psycopg2

# Connexion a la base PostgreSQL
connexion = psycopg2.connect(
    host="localhost",
    port=5432,
    database="trafic_aerien",
    user="postgres",
    password="VOTRE_MOT_DE_PASSE"    
)
cursor = connexion.cursor()

# Authentification OAuth2 (OpenSky Network)
payload = {
    "grant_type": "client_credentials",
    "client_id": "VOTRE_CLIENT_ID",
    "client_secret": "VOTRE_CLIENT_SECRET" 
}

reponse = requests.post(
    "https://auth.opensky-network.org/auth/realms/opensky-network/protocol/openid-connect/token",
    payload
)
data = reponse.json()
print("Token obtenu")

token = data["access_token"]
headers = {"Authorization": "Bearer " + token}

# Recuperation des etats de vol en temps reel
avions = requests.get("https://opensky-network.org/api/states/all", headers=headers)
avions = avions.json()

listes_vol = avions["states"]
print(len(listes_vol), "avions recuperes")

# Insertion de chaque vol dans la base
for vol in listes_vol:
    icao24 = vol[0]
    callsign = vol[1].strip() if vol[1] else None
    origin_country = vol[2]
    longitude = vol[5]
    latitude = vol[6]
    altitude = vol[7]
    on_ground = vol[8]
    velocity = vol[9]

    cursor.execute(
        """INSERT INTO releve_vols
           (icao24, callsign, origin_country, longitude, latitude, altitude, velocity, on_ground)
           VALUES (%s, %s, %s, %s, %s, %s, %s, %s)""",
        (icao24, callsign, origin_country, longitude, latitude, altitude, velocity, on_ground)
    )

cursor.execute("SELECT COUNT(*) FROM releve_vols")
print(cursor.fetchone())

connexion.commit()
cursor.close()
connexion.close()

Code SQL — Base & Analyses

Création de la table

CREATE TABLE releve_vols (
    id              SERIAL PRIMARY KEY,
    icao24          VARCHAR(6),
    callsign        VARCHAR(8),
    origin_country  VARCHAR(100),
    longitude       FLOAT,
    latitude        FLOAT,
    altitude        FLOAT,
    velocity        FLOAT,
    on_ground       BOOLEAN,
    horodatage      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Requêtes d'analyse

-- Top 10 des pays avec le plus d'avions
SELECT origin_country, COUNT(*) AS nombre_vols
FROM releve_vols
GROUP BY origin_country
ORDER BY nombre_vols DESC
LIMIT 10;

-- Nombre d'avions en vol vs au sol
SELECT on_ground, COUNT(*) AS nombre
FROM releve_vols
GROUP BY on_ground;

-- Vols dans une zone precise (bounding box autour de la France)
SELECT icao24, callsign, origin_country, longitude, latitude, altitude
FROM releve_vols
WHERE longitude BETWEEN -5 AND 10
  AND latitude  BETWEEN 41 AND 51
  AND on_ground = false;

-- Repartition des avions par zone (Europe, Amerique du Nord, Asie)
SELECT
    CASE
        WHEN longitude BETWEEN -5 AND 40  AND latitude BETWEEN 35 AND 70 THEN 'Europe'
        WHEN longitude BETWEEN -130 AND -60 AND latitude BETWEEN 25 AND 55 THEN 'Amerique du Nord'
        WHEN longitude BETWEEN 60 AND 150 AND latitude BETWEEN 10 AND 55 THEN 'Asie'
        ELSE 'Autre'
    END AS zone,
    COUNT(*) AS nombre_vols
FROM releve_vols
WHERE on_ground = false
GROUP BY zone
ORDER BY nombre_vols DESC;

-- Distribution des altitudes (tranches de 2000 m)
SELECT
    CASE
        WHEN altitude < 2000  THEN '0 - 2000 m'
        WHEN altitude < 4000  THEN '2000 - 4000 m'
        WHEN altitude < 6000  THEN '4000 - 6000 m'
        WHEN altitude < 8000  THEN '6000 - 8000 m'
        WHEN altitude < 10000 THEN '8000 - 10000 m'
        WHEN altitude < 12000 THEN '10000 - 12000 m'
        ELSE '12000+ m'
    END AS tranche_altitude,
    COUNT(*) AS nombre_vols
FROM releve_vols
WHERE on_ground = false AND altitude IS NOT NULL
GROUP BY tranche_altitude
ORDER BY tranche_altitude;

-- Statistiques generales du projet
SELECT
    COUNT(*)                                      AS total_releves,
    COUNT(DISTINCT icao24)                        AS avions_uniques,
    COUNT(DISTINCT origin_country)                AS pays,
    ROUND(AVG(altitude)::numeric, 2)              AS altitude_moyenne_m,
    ROUND(AVG(velocity * 3.6)::numeric, 2)        AS vitesse_moyenne_kmh,
    MIN(horodatage)                               AS premiere_collecte,
    MAX(horodatage)                               AS derniere_collecte
FROM releve_vols
WHERE on_ground = false;

-- Vue : statistiques par pays
CREATE OR REPLACE VIEW stats_par_pays AS
SELECT origin_country,
       COUNT(*)                                AS nombre_vols,
       ROUND(AVG(altitude)::numeric, 2)        AS altitude_moy_m,
       ROUND(AVG(velocity * 3.6)::numeric, 2)  AS vitesse_moy_kmh
FROM releve_vols
WHERE on_ground = false AND altitude IS NOT NULL
GROUP BY origin_country;

-- Vue : avions survolant la France
CREATE OR REPLACE VIEW vols_france AS
SELECT icao24, callsign, origin_country, longitude, latitude,
       ROUND(altitude::numeric, 2)             AS altitude_m,
       ROUND((velocity * 3.6)::numeric, 2)     AS vitesse_kmh,
       horodatage
FROM releve_vols
WHERE longitude BETWEEN -5 AND 10
  AND latitude  BETWEEN 41 AND 51
  AND on_ground = false;
